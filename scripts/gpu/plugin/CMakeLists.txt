cmake_minimum_required(VERSION 3.25)
project(RugbyIntegerFetch LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RELWITHDEBINFO_POSTFIX rd)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
find_package(rexglue CONFIG REQUIRED)
add_library(rexruntime ALIAS rex::runtime)
add_library(xxHash::xxhash ALIAS rex::xxhash)
set(REXGLUE_ROOT "${RUGBY_SDK_SOURCE}" CACHE PATH "ReXGlue 0.10.0 source directory")
set(REXGLUE_USE_D3D12 ON)
set(REXGLUE_USE_VULKAN OFF)
include("${REXGLUE_ROOT}/cmake/rex_pch.cmake")
add_compile_definitions(REX_PLATFORM_WINDOWS=1 NOMINMAX WIN32_LEAN_AND_MEAN)
add_compile_options(/clang:-fno-strict-aliasing /clang:-ffp-model=strict /clang:-fno-char8_t /clang:-msse4.1)
include_directories("${REXGLUE_ROOT}/include")
add_subdirectory("${REXGLUE_ROOT}/src/graphics" graphics)
